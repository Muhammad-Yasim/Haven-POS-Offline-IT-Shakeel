-- ============================================================================
-- SECTION 14: ROW LEVEL SECURITY (RLS)
-- ============================================================================
-- RLS provides database-level tenant isolation.
--
-- Flow:
--
--   Supabase Auth
--        │
--        ▼
--   auth.uid()
--        │
--        ▼
--   public.users.id
--        │
--        ▼
--   public.users.business_id
--        │
--        ▼
--   Current Business / Tenant
--
-- Every authenticated user can only access data belonging to their own
-- business_id.
--
-- IMPORTANT:
-- Registration of a brand-new business must be performed by your backend
-- using the Supabase service_role key because the user does not yet have a
-- public.users row from which current_business_id() can be determined.
--
-- The service_role key MUST NEVER be exposed to the frontend.
-- ============================================================================


-- ============================================================================
-- 1. HELPER FUNCTION: GET CURRENT USER'S BUSINESS ID
-- ============================================================================

CREATE OR REPLACE FUNCTION public.current_business_id()
RETURNS INT
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT business_id
  FROM public.users
  WHERE id = auth.uid()
  LIMIT 1;
$$;


-- Restrict who can execute this helper function.
REVOKE ALL ON FUNCTION public.current_business_id() FROM PUBLIC;

GRANT EXECUTE
ON FUNCTION public.current_business_id()
TO authenticated;

-- ============================================================================
-- MIGRATION: Register (cash drawer) support for POS Billing
-- ============================================================================
-- Adds:
--   - registers table (id, business_id, location_id, user_id, opened_at,
--     closed_at, opening_cash, closing_cash, status)
--   - register_id FK on sales, sell_returns, expenses
--   - RLS tenant isolation policy for registers, matching every other table
--
-- Deliberately NOT stored (calculated dynamically instead, see
-- src/lib/registerUtils.js and src/pages/registerReport.jsx):
--   total sales, cash sales, card sales, total refunds, total expenses,
--   expected cash, cash difference, product quantities sold.
-- ============================================================================

CREATE TYPE register_status_enum AS ENUM ('open', 'closed');

CREATE TABLE registers (
  id            INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  business_id   INT NOT NULL REFERENCES businesses(id) ON DELETE CASCADE,
  location_id   INT NOT NULL REFERENCES locations(id) ON DELETE RESTRICT,
  user_id       UUID NOT NULL REFERENCES users(id) ON DELETE RESTRICT,
  opened_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  closed_at     TIMESTAMPTZ,
  opening_cash  DECIMAL(12,2) NOT NULL DEFAULT 0 CHECK (opening_cash >= 0),
  closing_cash  DECIMAL(12,2) CHECK (closing_cash IS NULL OR closing_cash >= 0),
  status        register_status_enum NOT NULL DEFAULT 'open'
);

-- Only one OPEN register per location at a time. A location's drawer is
-- shared by whoever is on the POS terminal there — this mirrors how
-- posBilling.jsx already scopes stock/cart state to `locationId`, not to
-- the logged-in user. Remove this index if you'd rather allow one open
-- register per (location, user) instead.

CREATE INDEX idx_registers_business_location_status
  ON registers (business_id, location_id, status);

-- ----------------------------------------------------------------------------
-- Link transactions to the register that was open when they happened.
-- Nullable + ON DELETE SET NULL: transactions created outside of an open
-- register session (e.g. manual sales from the Sales module, or historical
-- rows from before this feature existed) simply have register_id = NULL and
-- are excluded from register reports, without breaking anything else.
-- ----------------------------------------------------------------------------

ALTER TABLE sales        ADD COLUMN register_id INT REFERENCES registers(id) ON DELETE SET NULL;
ALTER TABLE sell_returns ADD COLUMN register_id INT REFERENCES registers(id) ON DELETE SET NULL;
ALTER TABLE expenses     ADD COLUMN register_id INT REFERENCES registers(id) ON DELETE SET NULL;

CREATE INDEX idx_sales_register_id        ON sales(register_id);
CREATE INDEX idx_sell_returns_register_id ON sell_returns(register_id);
CREATE INDEX idx_expenses_register_id     ON expenses(register_id);

-- ----------------------------------------------------------------------------
-- RLS — identical tenant-isolation pattern used everywhere else in the schema
-- ----------------------------------------------------------------------------

ALTER TABLE registers ENABLE ROW LEVEL SECURITY;

CREATE POLICY tenant_isolation_registers
ON registers
FOR ALL
TO authenticated
USING (
  business_id = public.current_business_id()
)
WITH CHECK (
  business_id = public.current_business_id()
);

-- No changes needed to the existing tenant_isolation_sales /
-- tenant_isolation_sell_returns / tenant_isolation_expenses policies —
-- they already scope by business_id, and register_id is just an extra
-- nullable column on those same rows.






-- ============================================================================
-- MIGRATION: switch registers from one-open-per-location to
--            one-open-per-user
-- ============================================================================
-- Run this AFTER 001_add_registers.sql.
--
-- Before: a location could only have one open register, shared by whoever
-- was on that POS terminal.
-- After:  each USER can only have one open register at a time (at any
-- location), but a location can now have multiple open registers
-- simultaneously if multiple cashiers are each running their own till.
-- ============================================================================

DROP INDEX IF EXISTS uq_registers_one_open_per_location;

CREATE UNIQUE INDEX uq_registers_one_open_per_user
  ON registers (user_id)
  WHERE status = 'open';

-- Helpful for "does this user already have a register open" checks and for
-- the Active Register page.
CREATE INDEX IF NOT EXISTS idx_registers_business_user_status
  ON registers (business_id, user_id, status);

-- it can coexist with the per-user index above:
